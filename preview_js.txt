function openPreview() {
    let agendaContent = '';

    if (typeof tinymce !== 'undefined' && tinymce.get('agenda_details')) {
      agendaContent = tinymce.get('agenda_details').getContent();
    } else {
      const plain = document.getElementById('agenda_details').value;
      agendaContent = plain.trim()
        ? plain.replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;')
          .replace(/\n/g, '<br>')
        : '';
    }

    // Build full A4 HTML document for iframe
    const iframeDoc = `<!DOCTYPE html>
<html lang="en">
<head>
<meta charset="UTF-8">
<style>
  @import url('https://fonts.googleapis.com/css2?family=Calibri&display=swap');
  * { box-sizing: border-box; margin: 0; padding: 0; }
  html, body {
    width: 100%;
    height: 100%;
    font-family: Calibri, Arial, sans-serif;
    font-size: 12pt;
    line-height: 1.6;
    color: #000;
    background: white;
  }
  body {
    padding: 15mm 18mm;
    font-size: 13pt;
    line-height: 1.8;
  }
  p { margin: 0 0 6pt 0; }
  table { border-collapse: collapse; width: 100%; }
  td, th { padding: 4pt 6pt; }
  h1 { font-size: 18pt; font-weight: bold; margin-bottom: 8pt; }
  h2 { font-size: 14pt; font-weight: bold; margin-bottom: 6pt; }
  h3 { font-size: 12pt; font-weight: bold; margin-bottom: 4pt; }
  ul, ol { margin-left: 20pt; margin-bottom: 6pt; }
  blockquote { border-left: 3pt solid #2563eb; padding-left: 10pt; color: #4b5563; }
  a { color: #2563eb; }
  strong, b { font-weight: bold; }
  em, i { font-style: italic; }
  u { text-decoration: underline; }
</style>
</head>
<body>
${agendaContent || '<p style="color:#9ca3af;">(No agenda entered)</p>'}
</body>
</html>`;

    // Iframe mein inject karo
    const iframe = document.getElementById('preview-iframe');

    // Show modal first
    document.getElementById('previewModalOverlay').classList.add('active');
    document.body.style.overflow = 'hidden';

    // Iframe load hone ke baad height adjust karo
    setTimeout(() => {
      const iframeWin = iframe.contentWindow || iframe.contentDocument;
      const doc = iframe.contentDocument || iframeWin.document;
      doc.open();
      doc.write(iframeDoc);
      doc.close();

      // Content ki height ke mutabiq iframe resize karo
      setTimeout(() => {
        try {
          const body = doc.body;
          const html = doc.documentElement;
          const height = Math.max(
            body.scrollHeight, body.offsetHeight,
            html.clientHeight, html.scrollHeight, html.offsetHeight
          );
          iframe.style.height = Math.max(height, 297 * 3.7795) + 'px'; // min 297mm
          // preview-section bhi same height
          iframe.parentElement.style.minHeight = iframe.style.height;
        } catch (e) { }
      }, 100);
    }, 50);
  }

  function closePreview() {
    document.getElementById('previewModalOverlay').classList.remove('active');
    document.body.style.overflow = '';
  }

  document.getElementById('preview-btn').addEventListener('click', openPreview);
  document.getElementById('previewCloseBtn').addEventListener('click', closePreview);
  document.getElementById('previewCloseBtnFooter').addEventListener('click', closePreview);
  document.getElementById('previewModalOverlay').addEventListener('click', function (e) {
    if (e.target === this) closePreview();
  });


  document.getElementById('previewPrintBtn').addEventListener('click', function () {
    // TinyMCE content directly lo
    let agendaContent = '';
    if (typeof tinymce !== 'undefined' && tinymce.get('agenda_details')) {
      agendaContent = tinymce.get('agenda_details').getContent();
    } else {
      const plain = document.getElementById('agenda_details').value;
      agendaContent = plain.trim()
        ? plain.replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;').replace(/\n/g, '<br>')
        : '';
    }

    if (!agendaContent.trim()) {
      alert('Please enter some agenda content!');
      return;
    }

    // Naya popup window banao print ke liye
    const printWindow = window.open('', '_blank', 'width=900,height=700');

    printWindow.document.write(`<!DOCTYPE html>
<html lang="en">
<head>
<meta charset="UTF-8">
<title>Notesheet Print</title>
<style>

    line-height: 1.6;
    color: #000;
    background: white;
  }
  body { padding: 20mm; }
  @media print {
    body { padding: 0; }
    .no-print { display: none !important; }
  }
  p { margin: 0 0 6pt; }
  table { border-collapse: collapse; width: 100%; }
  td, th { padding: 4pt 6pt; }
  h1 { font-size: 18pt; font-weight: bold; margin-bottom: 8pt; }
  h2 { font-size: 14pt; font-weight: bold; margin-bottom: 6pt; }
  h3 { font-size: 12pt; font-weight: bold; margin-bottom: 4pt; }
  ul, ol { margin-left: 20pt; margin-bottom: 6pt; }
  blockquote { border-left: 3pt solid #2563eb; padding-left: 10pt; color: #4b5563; }
  strong, b { font-weight: bold; }
  em, i { font-style: italic; }
  u { text-decoration: underline; }

  /* Print button - screen only */
  .print-toolbar {
    position: fixed;
    top: 10px;
    right: 10px;
    display: flex;
    gap: 10px;
    z-index: 9999;
  }
  .btn-do-print {
    padding: 10px 20px;
    background: #1e40af;
    color: white;
    border: none;
    border-radius: 8px;
    font-size: 14px;
    font-weight: 600;
    cursor: pointer;
  }
  .btn-do-print:hover { background: #1e3a8a; }
  .btn-do-close {
    padding: 10px 20px;
    background: #6b7280;
    color: white;
    border: none;
    border-radius: 8px;
    font-size: 14px;
    font-weight: 600;
    cursor: pointer;
  }
</style>
</head>
<body>
  <div class="print-toolbar no-print">
    <button class="btn-do-print" onclick="window.print()">🖨️ Print / Save as PDF</button>
    <button class="btn-do-close" onclick="window.close()">✕ Close</button>
  </div>
  <div class="letter-content">
    ${agendaContent}
  </div>
  <script>
    // Auto print dialog open karo
    window.onload = function() {
      setTimeout(function() {
        window.print();
      }, 500);
    };
  <\/script>
</body>
</html>`);

    printWindow.document.close();
  });
