          tools: { title: 'Tools', items: 'spellchecker spellcheckerlanguage | a11ycheck code wordcount' },
          table: { title: 'Table', items: 'inserttable | cell row column | advtablesort | tableprops deletetable' },
        },
        height: 460,
        min_height: 600,
        autoresize_bottom_margin: 50,
        resize: true,
        plugins: [
          'advlist', 'autolink', 'lists', 'link', 'image', 'charmap',
          'searchreplace', 'visualblocks', 'code', 'fullscreen',
          'insertdatetime', 'media', 'table', 'wordcount', 'codesample',
          'emoticons', 'nonbreaking', 'pagebreak', 'quickbars', 'autoresize'
        ],
        toolbar_mode: 'wrap',
        paste_data_images: true,
        paste_as_text: false,
        /* ── ✂️ AUTO-CLEAN WORD PASTE ── */
        paste_postprocess: function (plugin, args) {
          var node = args.node;

          /* 1 ▸ Strip all mso-* / Word-only CSS properties */
          node.querySelectorAll('[style]').forEach(function (el) {
            var raw = el.getAttribute('style') || '';
            var cleaned = raw.split(';')
              .map(function (s) { return s.trim(); })
              .filter(function (s) {
                if (!s) return false;
                var prop = s.split(':')[0].trim().toLowerCase();
                return !prop.startsWith('mso-')
                  && prop !== 'tab-stops'
                  && prop !== 'layout-grid-mode'
                  && prop !== 'punctuation-wrap'
                  && prop !== 'text-autospace';
              })
              .join('; ');
            if (cleaned) { el.setAttribute('style', cleaned); }
            else { el.removeAttribute('style'); }
          });

          /* 2 ▸ Remove Word-generated class attributes */
          node.querySelectorAll('[class]').forEach(function (el) {
            el.removeAttribute('class');
          });

          /* 3 ▸ Normalise paragraphs → Calibri 12pt, line-height 1.8 */
          node.querySelectorAll('p').forEach(function (p) {
            p.style.fontFamily = 'Calibri, Arial, sans-serif';
            p.style.fontSize = '12pt';
            p.style.lineHeight = '1.8';
            if (!p.style.margin) p.style.margin = '0 0 6pt 0';
          });

          /* 4 ▸ Normalise headings */
          var headingSizes = { H1: '22pt', H2: '18pt', H3: '14pt', H4: '13pt' };
          Object.keys(headingSizes).forEach(function (tag) {
            node.querySelectorAll(tag).forEach(function (h) {
              h.style.fontFamily = 'Calibri, Arial, sans-serif';
              h.style.fontSize = headingSizes[tag];
              h.style.lineHeight = '1.4';
              h.style.fontWeight = 'bold';
            });
          });

          /* 5 ▸ Normalise inline spans */
          node.querySelectorAll('span').forEach(function (span) {
            if (!span.style.fontFamily)
              span.style.fontFamily = 'Calibri, Arial, sans-serif';
            if (!span.style.fontSize)
              span.style.fontSize = '12pt';
          });

          /* 6 ▸ Fix tables → professional notesheet style */
          node.querySelectorAll('table').forEach(function (tbl) {
            tbl.style.borderCollapse = 'collapse';
            tbl.style.width = '100%';
            tbl.style.margin = '8pt 0';
            tbl.removeAttribute('width');
            tbl.removeAttribute('border');
            tbl.removeAttribute('cellspacing');
            tbl.removeAttribute('cellpadding');
          });
          node.querySelectorAll('td, th').forEach(function (cell) {
            cell.style.border = '1px solid #d1d5db';
            cell.style.padding = '6px 10px';
            cell.style.fontFamily = 'Calibri, Arial, sans-serif';
            cell.style.fontSize = '12pt';
            cell.style.lineHeight = '1.6';
          });
          node.querySelectorAll('th').forEach(function (th) {
            th.style.background = '#d1fae5';
            th.style.fontWeight = '600';
          });

          /* 7 ▸ Collapse consecutive empty paragraphs (max 1 blank line) */
          var paras = Array.from(node.querySelectorAll('p'));
          var lastEmpty = false;
          paras.forEach(function (p) {
            var isEmpty = p.textContent.trim() === '' ||
              p.innerHTML.trim() === '&nbsp;' ||
              p.innerHTML.trim() === '<br>';
            if (isEmpty && lastEmpty) {
              p.parentNode && p.parentNode.removeChild(p);
            } else {
              lastEmpty = isEmpty;
            }
          });

          /* 8 ▸ Remove Word conditional-comment leftovers (w:, o:, v: tags) */
          ['w\\:*', 'o\\:*', 'v\\:*'].forEach(function (sel) {
            try {
              node.querySelectorAll(sel).forEach(function (el) { el.remove(); });
            } catch (e) { }
          });
        },
